#### D. Deploy Get-Quote Edge Function

**File**: `get-quote/index.ts` (deployed via Supabase MCP)

```typescript
import { serve } from 'https://deno.land/std@0.168.0/http/server.ts'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    const { context } = await req.json()

    if (!context || !['preWorkout', 'workout', 'completion'].includes(context)) {
      return new Response(
        JSON.stringify({ error: 'Invalid context' }),
        { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    const supabaseClient = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_ANON_KEY') ?? '',
    )

    // Fetch all active quotes for the context
    const { data: quotes, error } = await supabaseClient
      .from('workout_quotes')
      .select('quote_text')
      .eq('context', context)
      .eq('is_active', true)

    if (error) throw error
    if (!quotes || quotes.length === 0) {
      return new Response(
        JSON.stringify({ error: 'No quotes found' }),
        { status: 404, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    // Return random quote
    const randomQuote = quotes[Math.floor(Math.random() * quotes.length)]

    return new Response(
      JSON.stringify({ quote: randomQuote.quote_text }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  } catch (error) {
    return new Response(
      JSON.stringify({ error: error.message }),
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  }
})
```

### 2. Client Code Changes (OTA)

#### A. Create Quotes Client (`src/lib/quotesClient.js`)

```javascript
import { supabase } from './supabase';
import { getQuote as getLocalQuote } from '../utils/quotes';

// In-memory cache (1 hour TTL)
const cache = {
  preWorkout: { quote: null, timestamp: 0 },
  workout: { quote: null, timestamp: 0 },
  completion: { quote: null, timestamp: 0 }
};

const CACHE_TTL = 60 * 60 * 1000; // 1 hour in milliseconds
const TIMEOUT_MS = 1000; // 1 second timeout for server fetch

/**
 * Fetch quote from server with timeout and fallback
 * @param {string} context - 'preWorkout', 'workout', or 'completion'
 * @returns {Promise<string>} - Quote text
 */
export const getQuoteFromServer = async (context) => {
  // Check cache first
  const now = Date.now();
  const cached = cache[context];
  if (cached.quote && (now - cached.timestamp) < CACHE_TTL) {
    return cached.quote;
  }

  try {
    // Create timeout promise
    const timeoutPromise = new Promise((_, reject) =>
      setTimeout(() => reject(new Error('Timeout')), TIMEOUT_MS)
    );

    // Create fetch promise
    const fetchPromise = supabase.functions.invoke('get-quote', {
      body: { context }
    });

    // Race timeout vs fetch
    const { data, error } = await Promise.race([fetchPromise, timeoutPromise]);

    if (error) throw error;
    if (!data || !data.quote) throw new Error('No quote in response');

    // Update cache
    cache[context] = { quote: data.quote, timestamp: now };
    return data.quote;

  } catch (error) {
    console.log(`Quote server fetch failed (${context}), using local fallback:`, error.message);
    // Fallback to local quotes
    return getLocalQuote(context);
  }
};
```

## Data Model Clarity

### `workout_type` vs `pattern`
- **`workout_type`**: Enum stored in database (`'max_test'` OR `'volume'`)
  - Represents the workout category
  - Max test = single set to failure
  - Volume = 8 sets with planned reps

- **`pattern`**: Text stored in database (e.g., "Equal Sets", "Pyramid", "Descending")
  - Represents the rep distribution pattern for volume workouts
  - Only applies to `workout_type = 'volume'`
  - Max test workouts have `pattern = null`

### Plan Tuple Fields
- **`pattern`**: Pattern name from route params (e.g., "Equal Sets", "Pyramid")
- **`cycle_num`**: Cycle number from Dashboard RPC (1, 2, 3, ...)
- **`workout_num`**: Workout number within cycle from route params (1-8)
